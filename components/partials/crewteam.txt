"use client";

import apiClient from "@/app/axios";
import Image from "next/image";
import Link from "next/link";
import { useEffect, useState } from "react";
import LoadingAnimation from "./loadingAnimation";

export default function CrewTeam() {
  const [users, setUsers] = useState([]);

  useEffect(() => {
    const fetchUsers = async () => {
      try {
        const response = await apiClient.get(`/api/user`);
        if (response) {
          setUsers(response?.data?.data || []);
        } else {
          <LoadingAnimation isLoading={true} />;
        }
      } catch (error) {
      }
    };

    fetchUsers();
  }, []);

  return (
    <section className="grid grid-cols-1 lg:grid-cols-3 gap-[32px] relative">
      {users.length > 0 ? (
        users.map((user, index) => (
          <div key={index} className="h-[400px] relative group shadow-2xl">
            <Image
              src={`${user?.profile}` || "/images/default.webp"}
              width={300}
              height={300}
              alt={user.firstname || "Profile"}
              className="w-full h-full object-cover"
            />
            <span className="absolute inset-0 text-white flex items-center justify-between bg-black/90 px-[16px] transition-all duration-500 opacity-0 group-hover:opacity-100 h-0 group-hover:h-[100px] transform">
              <h3 className="text-2xl uppercase">{user?.firstname}</h3>
              <Link href={`/coach/${user.id}`} className="pointer">
                <svg
                  fill="none"
                  viewBox="0 0 24 24"
                  strokeWidth="1.5"
                  stroke="currentColor"
                  className="size-8"
                >
                  <path
                    strokeLinecap="round"
                    strokeLinejoin="round"
                    d="m11.25 11.25.041-.02a.75.75 0 0 1 1.063.852l-.708 2.836a.75.75 0 0 0 1.063.853l.041-.021M21 12a9 9 0 1 1-18 0 9 9 0 0 1 18 0Zm-9-3.75h.008v.008H12V8.25Z"
                  />
                </svg>
              </Link>
            </span>
          </div>
        ))
      ) : (
        <div className="absolute inset-0 items-center center">
          <LoadingAnimation isLoading={true} />
        </div>
      )}
    </section>
  );
}
